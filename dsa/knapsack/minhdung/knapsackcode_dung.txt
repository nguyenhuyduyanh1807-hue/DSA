#include <iostream>
using namespace std;

const int N = 4;

int weights[N] = {2, 3, 4, 5};
int values[N] = {3, 4, 5, 6};

int capacity = 5;

int bestValue = 0;

// Simple bound function
int calculateBound(int index, int currentValue) {

    int remainingValue = 0;

    for (int i = index; i < N; i++) {
        remainingValue += values[i];
    }

    return currentValue + remainingValue;
}

void knapsack(int index,
              int currentWeight,
              int currentValue) {

    // Invalid case
    if (currentWeight > capacity) {
        return;
    }

    // Update best answer
    if (currentValue > bestValue) {
        bestValue = currentValue;
    }

    // End of items
    if (index == N) {
        return;
    }

    // Branch and Bound
    int bound = calculateBound(index, currentValue);

    if (bound <= bestValue) {
        return;
    }

    // TAKE item
    knapsack(index + 1,
             currentWeight + weights[index],
             currentValue + values[index]);

    // SKIP item
    knapsack(index + 1,
             currentWeight,
             currentValue);
}

int main() {

    knapsack(0, 0, 0);

    cout << "Best value = "
         << bestValue << endl;

    return 0;
}
