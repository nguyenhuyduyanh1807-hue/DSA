#include <iostream>
#include<vector>
using namespace std;

// Recursive function to place queens
void placeQueens(int i, vector<int> &cols, vector<int> &leftDiagonal, 
                     vector<int> &rightDiagonal, vector<int> &cur, 
                     vector<vector<int> > &result) {
    int n = cols.size();
    /*- cái cols là hàm quan lý column. If you wanna place a queen in j column, mark cols[j] = 1. The next queen will scan if the first queen is equal j or not.
	  - duong chéo xuôi (from left to down right) (dòng 29)
	  - duong chéo nguoc (from right to down left) (dòng 28)
	  - hàm cur là current queen, báo location cua các queen dang available cho queen sau và là hàm luu các queen da hop le 
	  - result: tap hop all cur de ra ket qua cuoi cùng
	  - cols size là kích thuoc bàn co based on do dài cols dc truyen vào*/
  
    // base case: If all queens are placed
    if(i == n) { /*i cho biet mình dang o hàng nào trong bàn co. nên ví du a chess board có n = 4 (4 hàng 4 cols) thì i = 0 là hàng 1,... den i = 3 (hàng 4). và khi i = 4 thì nó realise nó da hoàn thành nhiem vu (vì 4 hàng truoc xong r)*/
        result.push_back(cur); /*bê các vi trí hien tai cua queen dã save trong cur và cât vào kho result*/
        return;
    }

    // Consider the row and try placing
    // queen in all columns one by one
    for(int j = 0; j < n; j++){ /*xét hàng hien tai i, consider queen vào cols j tu 0 den n - 1 xem có hop le không*/

        // Check if the queen can be placed
        if(cols[j] || rightDiagonal[i + j] || /*giong duong chéo xuôi thì duong chéo nguoc cung có công thuc chung là hàng + cols*/
                        leftDiagonal[i - j + n - 1]) /*công thuc cua duong chéo xuôi là hàng - columns. Vì có kha nang âm (hàng 0 cols 2) nên ph + thêm n - 1*/
            continue;
        /* ý nghia || là hoac. Chi cân 1 trong 3 giá tri thoa mãn thì có nghia that place có queen và se skip luôn sang cols moi*/
        
        // if the queen is placed, continue to this step
        cols[j] = 1;
        rightDiagonal[i+j] = 1; 
        leftDiagonal[i - j + n - 1] = 1;
        cur.push_back(j+1);
        /*cho con queen sô do tai vi tri này. ghi nhân vi trí cua queen*/

        placeQueens(i + 1, cols, leftDiagonal, rightDiagonal, cur, result);
        /*goi lai ham placequeens o trên (dòng 6), i + 1: tang hàng lên dê di xuông hàng tiêp theo*/

        // remove the queen from current cell /*khi di xuông hàng duoi và gap dead end, dây là hàm kêt thúc và xóa queen*/
        cur.pop_back(); /*xóa queen*/
        cols[j] = 0;
        rightDiagonal[i+j] = 0; 
        leftDiagonal[i - j + n - 1] = 0;      
        /*= 0 dê reset vi trí */
    }
}

// Function to find the solution
// to the N-Queens problem
vector<vector<int> > nQueen(int n) { //hàm khoi tao (nhân kich thuoc bàn co, câp phát memory storage cho các mang)

    // array to mark the occupied cells
    vector<int> cols(n, 0);
    vector<int> leftDiagonal(n*2, 0); //khai báo n*2 vì công thuc gôc là hàng + côt nên nó se nhân dôi => n*2 dê dam bao du bô nho tránh tràn mang hoac thieu storage 
    vector<int> rightDiagonal(n*2, 0); 
    vector<int> cur; //ghi nhân dap án tam thoi
    vector<vector<int> > result; //gom dap án cuôi cùng

    // Place queens
    placeQueens(0, cols, leftDiagonal, rightDiagonal, cur, result); 

    return result;
}
// Run the program 
int main() {
    int n = 4; /*thiet lap matrix (in this example là matrix 4x4)*/
    vector<vector<int> > ans = nQueen(n); /*biên ans dê rút gon bieu thuc*/
	for (int i = 0; i < ans.size(); i++) { /*duyet qua dap an*/
	    cout << "--- ANSWER " << i + 1 << " ---" << endl; /*dap an thu 1, 2, 3...*/
	    
	    cout << "       ";
        for(int c = 1; c <= n; c++) cout << "Col " << c << "  ";
        cout << endl; /*ve tiêu de (cols)*/
        
        for (int r = 0; r < n; r++) { 
            cout << "Row " << r + 1 << "  "; /*ve tiêu de (rows)*/
            int queen_col_index = ans[i][r] - 1; 
			/*luu vi trí cols tai hàng r (o answer thu i). 
			  VD: ans[i][0]=2 thi no se la hang 1 cols 2.
			  - 1 vì máy tính run tu index 0 (0, 1, 2, 3) còn o trên van dang save theo (1, 2, 3, 4)*/
            for (int c = 0; c < n; c++) { /*ve ra neu queen o day thì Q còn không thi .*/
                if (c == queen_col_index) {
                    cout << "[Q]    "; 
                } else {
                    cout << "[.]    "; 
                }
            }
            cout << endl << endl; //xuong dòng sau khi xong 1 rows
        }
        cout << endl; //xuong dòng sau khi xong toàn bo bàn co
    }
    return 0;
}
