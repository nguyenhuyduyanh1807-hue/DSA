#include <iostream>
#include <vector>
#include <cmath>
using namespace std;

void printSolution (const vector<int>& board, int n) { //board holds the column position for each row.
	for (int i = 0; i < n; i++){
		for (int j = 0; j < n; j++) {
			if (board [i] == j) // if the queen in i-th row is put in the j-th column.
				cout << "Q" ;
			else 
				cout << ".";
		}
		cout << endl;
	}
	cout << endl;
}

bool isSafe(const vector <int>& board, int row, int col ) {
	for ( int i = 0; i < row; i++){
		if (board [i] == col) return false; //check for column repetition
		if (abs(board[i]-col) == abs (i-row)) return false; 
	}
	return true;
}

void solveNQueens1 (vector <int> &board, int row, int n) {
	if (row == n) {
		printSolution (board, n);
		return;
	}
	for (int col = 0; col < n; col++){
		if (isSafe(board, row, col)) {
			board[row ] = col; //place the queen in here.
			solveNQueens1 (board, row +1,n); // call for the next row. 
		}
	}
}

void solveNQueens2 (int n ){
	vector <int> board (n,-1); //create a board of size n*n with no queens on it. 
	solveNQueens1 (board, 0, n); // start backtracking.
}

int main() {
	int n;
	cout << "Enter N : ";
	cin >> n;
	solveNQueens2 (n);
	return 0;
}z
